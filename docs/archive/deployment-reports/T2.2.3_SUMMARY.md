### Document Upload Component
- **Drag-and-Drop**: Visual feedback for drag operations
- **File Validation**: Type (PDF, PNG, JPG) and size (10MB) checks

- Returns `source_url` for uploaded files

## Usage Example

```tsx
import { DocumentUpload } from '@/components/DocumentUpload';

function VendorPortal() {
  const handleUpload = async (url: string) => {
    // Save document URL to backend
    await vendorService.updateVendorRequest(token, {
      document_url: url
    });
  };

  return (
    <DocumentUpload
      onUpload={handleUpload}
      currentUrl={vendorRequest?.document_url}
      disabled={isUploading}
    />
  );
}
```
